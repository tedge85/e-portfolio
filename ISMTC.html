def build_tree(data, parent=None): # Microsoft Copilot (2025)
    """
        Builds tree recursively from root node.

        Args:
            data (dict): Node labels, detailing root, parents and children.

        Returns:
            Node: The nodes of the tree constructed.

    """

    # Create a node for the current dictionary entry (Microsoft Copilot, 2025).
    node = Node(data["a"], parent=parent)

    # Check if this node has children and recursively process them (Microsoft Copilot, 2025).
    if "children" in data:
        for child in data["children"]:
            build_tree(child, parent=node)

    return node


def load_json_file(file_name):
    """
        Loads a selected JSON file into a dictionary structure.

        Args:
            file_name (string): Name assigned to JSON file.

        Returns:
            dict: A dictionary form of the JSON file or None if not found.

    """
    try:
        # Open file (in read mode) with '.json' if not specified in file_name.
        if not file_name.endswith(".json"):
            with open(f"{file_name}.json", "r", encoding="utf-8") as f:
                return json.load(f)
        else:
            # Do not add '.json' if already specified in file_name.
            with open(f"{file_name}", "r", encoding="utf-8") as f:
                return json.load(f)

    except FileNotFoundError:
        return None


def list_leaf_nodes(root):
    """
        Lists all leaf node names.

        Args:
            root (Node): the root node object of the tree constructed.

        Returns:
            Str: A string of all leaf node names, separated by new lines.

    """

    node_str = ""

    # Iterate through all nodes and extract name attribute, adding it to node_str
    # with new line and tab separating each string.
    for node in PreOrderIter(root):
        if node.is_leaf:
            node_str += node.name
            node_str += "\n\t"
            node_str += " "

    return node_str


def node_exists(node_name, root):
    """
        Confirms whether the given leaf node name is present in the tree.

        Args:
            node_name (str): User-inputted name to match to node attributes.
            root (Node): the root node object of the tree constructed.

        Returns:
            bool: True if the node exists, otherwise False.

    """
    # Iterate through all nodes to find a name attribute that matches user's inputted name.
    for node in PreOrderIter(root):
        if node.is_leaf:
            if node.name.lower() == node_name.lower():
                return True

    return False


def return_matching_node(node_name, root):
    """
        Retrieves a leaf node that matches the given name.

        Args:
            node_name (str): User-inputted name to match to node attributes.
            root (Node): the root node object of the tree constructed.

        Returns:
            optional[Node]: Node object if found, None otherwise.

    """
    # Iterate through all nodes to find a name attribute that matches user's inputted name.
    for node in PreOrderIter(root):
        if node.is_leaf:
            if node.name.lower() == node_name.lower():

                return node

    return None


def convert_to_float(input_value):
    """
        Attempts to convert an integer to a float.

        Args:
            input_value (int): User-inputted value.

        Returns:
            Optional[float]: The converted input_value if successful, otherwise None.

    """
    try:
        return float(input_value)

    except ValueError:

        return None


def assign_threat_value(target_node, threat_value, root):
    """
        Assigns the given value to a leaf node that matches the given name.

        Args:
            target_node (str): Name of the leaf node to update.
            threat_value (float): Value to assign to threat_value attribute.
            root (Node): root Node object of tree constructed.

        Returns:
            None.

    """
    # Iterate through all leaf nodes to find a name attribute that matches to known node.
    for node in PreOrderIter(root): # c0fec0de(2023)
        if node.is_leaf:
            if node == target_node:
                node.threat_value = round(threat_value, 1) # When found, assign inputted
                                                        # threat value to threat_value attribute.


def agregate_threat_scores(root):
    """
        Adds threat_value attributes for all leaf nodes in the tree.

        Args:
            root (Node): The root node of tree constructed.

        Returns:
            float: The sum of all threat values, rounded to one decimal place.

    """
    total_score = float(0)

    # Iterate through all nodes to see if they have a threat_value assigned. If they do,
    # add the score to total_score varaible. If not, ignore.
    for node in PreOrderIter(root):
        if node.is_leaf:
            if hasattr(node, "threat_value"):
                total_score += float(node.threat_value)

    return round(total_score, 1)


def print_threat_attributes(root):
    """
        Prints all threat values assigned to leaf nodes.

        Args:
            root (Node): The root node of tree constructed.

        Returns:
            None.

    """
    # Iterate through all leaf nodes to see if they have a threat_value assigned. If they do,
    # print node name and value within string interpolation sentence, otherwise print
    # node name and 'has no threat value assigned'..
    for node in PreOrderIter(root):
        if node.is_leaf:
            if hasattr(node, "threat_value"):
                print(f"Leaf node '{node.name}' has threat probability value: {node.threat_value}")
            else:
                print(f"Leaf node '{node.name}' has no threat probability value asssigned")


def export_tree_to_dot_file(root, file_name): # OpenAI ChatGPT-4 (2025)
    """
        Exports tree constructed to dot file, ready to be converted to image.

        Args:
            root (Node): The root node of tree constructed.
            file_name (str): The name of the file to be saved.

        Returns:
            None.

    """
    DotExporter(root).to_dotfile(f"{file_name}.dot") # Pass the nodes of the tree to the Dotxcporter
                                             # method that outputs the nodes to a dot file that can
                                             # be processed by image_output_confirmed() to output a

        Args:
            path (str): The directory path specifying the location.

        Returns:
            bool: True if the directory exists, otherwise False

def combine_file_paths(directory, file_name):
    """
        Combines a directory path with a file name to make a full file path.

        Args:
            directory (str): The directory path specifying the location.
            file_name (str): The name of the file.

        Returns:
            str: The full file path, formatted based on the operating system.

    """

    return os.path.join(directory, file_name)


def output_image(directory, file_name): # OpenAI ChatGPT-4 (2025)
    """
        Processes tree data in dot file and outputs graphical PNG file representation of tree.

        Args:
            directory (str): User-inputted folder location.
            file_name (str): The name of the file to be saved.

        Returns:
            bool: True if image outputted successfully, otherwise False.

    """
    ## The below processes were chosen instead of the more efficient to_picture
    # method (C0feC0de, 2023) as this kept resulting in write permission denial.


    # Read dot file of nodes and output as PNG file in inputted path.
    try:
        subprocess.check_call([
            "dot",
            f"{file_name}.dot",
            "-Tpng",
            "-o",
            combine_file_paths(directory, file_name) + ".png"
        ])
        return True

    # If file cannot be processed, catch the error.
    except subprocess.CalledProcessError:
        return False


def record_threat_values_for_all_nodes(root):
    """
        Records threat values for all leaf nodes of the tree.

        Args:
            root (Node): The root node of tree constructed.

        Returns:
            list: List of dictionaries detailing node name and threat values.

    """
    threat_list = []

    # Iterate through all leaf nodes, extracting name attribute and assigning as value for 'name'
    # key.
    for node in PreOrderIter(root):
        if node.is_leaf:
            node_data = {"name": node.name}

            # If node has a threat value attribute assigned, assign this as value for 'threat_value'
            # key, otherwise assign '0'.
            if hasattr(node, "threat_value"):
                node_data["threat value"] = node.threat_value
            else:
                node_data["threat value"] = 0

            threat_list.append(node_data)

    return threat_list


def output_threat_values_to_txt_file(file_name, list_of_dicts, file_path, root):
    """
        Outputs threat values for all leaf nodes to a text file.

        Args:
            list_of_dicts (list): List of dictionaries containing node names and threat values.
            file_path (str): User-inputted path for file.
            root (Node): The root node of tree constructed.

        Returns:
            None.

    """
    with open(file_path, "w", encoding="utf-8") as f:

         # Title for outputted text file.
        str_to_print = ["\t\t\t\t\t\tTHREAT PROBABILITY VALUES: " + file_name]
        str_to_print.append("\n\t\t\t\t\t\t----------------------------------------------")

        # Iterate through each dictionary in list_of_dicts and include name and threat_value
        # attributes in string interpolation, separating each nodes detail with a new line.
        for line in list_of_dicts:
            tpv = "THREAT PROBABILITY VALUE: " # Variable used to avoid overly long line (356).
            str_attrs = [
                    "\n\nLEAF  NODE:\t\t\t\t\t\t\t\t\t\t\t\t" + tpv + str(line["threat value"]),
                    "\n" + line["name"]
                    ]
            str_to_print.append(" ".join(str_attrs))

        # Include threat value total at bottom of text file.
        total_threat_score = str(agregate_threat_scores(root))

        otar = "OVERALL THREAT ASSESSMENT RATING: " # Variable used to avoid overly long line (366).
        str_to_print.append("\n\t\t\t\t\t\t\t\t\t\t\t\t\t--------------------------------------")
        str_to_print.append("\n\t\t\t\t\t\t\t\t\t\t\t\t\t" + otar + total_threat_score)
        str_to_print.append("\n\t\t\t\t\t\t\t\t\t\t\t\t\t--------------------------------------")

        f.write("\n".join(str_to_print))


def choice_confirmed(choice):
    """
        checks whether user has inputted 'Y' or 'y'.

        Args:
            choice (str): user-inputted string to evaluate.

        Returns:
            bool: True if 'Y' or 'y' inputted, otherwise False.

    """
    if choice.lower() == "y":
        return True

    return False
