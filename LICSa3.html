-- 1. List all Employees whose salary is greater than 1,000 but not 2,000. Show the Employee Name, Department and Salary.
SELECT ENAME AS 'EMPLOYEE NAME', DNAME AS 'DEPARTMENT', SAL AS 'SALARY'
FROM DEPT, EMP
WHERE SAL > 1000 AND SAL <= 2000;

-- 2. Count the number of people in department 30 who receive a salary and a commission. 
SELECT COUNT(*)
FROM EMP
WHERE SAL IS NOT NULL AND SAL != 0 AND COMM IS NOT NULL AND COMM != 0 AND DEPTNO = 30;

-- 3. Find the name and salary of the employees that have a salary greater or equal to 1,000 and live in Dallas.
SELECT ENAME AS 'DALLAS EMPLOYEE NAME', SAL AS 'SALARY'
FROM EMP AS E
INNER JOIN DEPT AS D ON E.DEPTNO = D.DEPTNO 
WHERE E.SAL >= 1000 AND D.LOC = 'DALLAS';

-- 4. Find all departments that do not have any current employees. (4 marks)
SELECT E.*
FROM EMP AS E
WHERE NOT EXISTS (
    SELECT 1
    FROM DEPT AS D
    WHERE D.DEPTNO = E.DEPTNO
      AND D.DNAME = 'ACCOUNTING'
      OR D.DNAME = 'RESEARCH'
      OR D.DNAME = 'SALES'
);

-- 5. List the department number, the average salary, and the number/count of employees of each department.
SELECT D.DNAME AS 'DEPARTMENT NAME', E.DEPTNO AS 'DEPARTMENT NUMBER', AVG(E.SAL) AS 'AVERAGE SALARY', COUNT(E.EMPNO) AS 'NUMBER OF EMPLOYEES'
FROM DEPT AS D
INNER JOIN EMP AS E ON E.DEPTNO = D.DEPTNO
GROUP BY D.DEPTNO;
